    🎯 第一层：概念与背景

    1.1 为什么需要这些工具类型

    问题场景 传统解决方案 问题 现代方案
    存储多种类型之一 union ❌ 类型不安全、无法存储非POD类型 ✅ std::variant
    表示"可能无值" 指针、特殊值（-1, nullptr） ❌ 语义不清晰、容易误用 ✅ std::optional
    存储任意类型 void*、继承多态 ❌ 丢失类型信息、需要手动管理 ✅ std::any
    存储可调用对象 函数指针、仿函数 ❌ 无法统一处理、难以传递状态 ✅ std::function 
    // ❌ 传统union：类型不安全
union Data {
    int i;
    double d;
};
Data data;
data.i = 42;
std::cout << data.d;  // ❌ 未定义行为

// ✅ 使用std::variant：类型安全
std::variant<int, double> v = 42;
std::cout << std::get<int>(v);  // ✅ 安全

    1.2 四大工具类型的定位

    std::variant<Types...> (C++17)
    📦 类型安全的标签联合（Tagged Union）
    🎯 存储多种固定类型之一
    ⚡ 零开销抽象，无堆分配
    💡 状态机、AST节点、多态数据
    std::optional<T> (C++17)
    📦 表示"可能存在的值"
    🎯 替代指针/特殊值表示"无值"
    ⚡ 栈上分配，sizeof(T) + 1 byte
    💡 函数返回值、可选配置
    std::any (C++17)
    📦 类型擦除的万能容器
    🎯 存储任意类型
    ⚡ 可能堆分配
    💡 插件系统、配置存储
    std::function<Signature> (C++11)
    📦 统一的可调用对象包装器
    🎯 存储任何可调用对象
    ⚡ 虚函数调用，可能堆分配
    💡 回调、事件系统、策略模式

    ⚙️ 第二层：核心机制与实现原理

    2.1 std::variant - 类型安全的联合体

    基本用法

    #include <variant>

std::variant<int, double, std::string> v;

// 赋值
v = 42;           // int
v = 3.14;         // double
v = "hello";      // string

// 访问：std::get
auto s = std::get<std::string>(v);  // OK
// auto i = std::get<int>(v);        // 抛出异常

// 安全访问：std::get_if
if (auto p = std::get_if<std::string>(&v)) {
    std::cout << *p;
}

// 检查类型
std::cout << v.index();  // 当前类型索引
if (std::holds_alternative<std::string>(v)) {
    // ...
}

    std::visit - 模式匹配

    // 访问器
std::visit([](auto&& arg) {
    std::cout << arg << "\n";
}, v);

// C++20 overloaded pattern
template<class... Ts> struct overloaded : Ts... { 
    using Ts::operator()...; 
};
template<class... Ts> overloaded(Ts...) -> overloaded<Ts...>;

std::visit(overloaded{
    [](int i) { std::cout << "int: " << i; },
    [](double d) { std::cout << "double: " << d; },
    [](const std::string& s) { std::cout << "string: " << s; }
}, v);

    内部实现

    // 简化实现概念
template<typename... Types>
class variant {
    std::aligned_union_t<0, Types...> storage;
    size_t type_index;  // 类型标签
};
    特性 说明
    大小 max(sizeof(Types...)) + sizeof(size_t)
    类型安全 运行时检查，访问错误类型抛异常
    性能 零开销，无堆分配
    异常安全 valueless_by_exception状态

    2.2 std::optional - 可选值语义

    基本用法

    #include <optional>

// 创建
std::optional<int> opt1;              // 空
std::optional<int> opt2 = 42;         // 有值
std::optional<int> opt3 = std::nullopt;  // 显式空

// 检查
if (opt2) {  // 或 opt2.has_value()
    std::cout << *opt2;
}

// 访问
int v1 = *opt2;              // 不检查
int v2 = opt2.value();       // 检查，抛异常
int v3 = opt2.value_or(0);  // 默认值

// 修改
opt2 = 100;
opt2.emplace(200);
opt2.reset();

    函数返回场景

    std::optional<int> parse(const std::string& s) {
    try {
        return std::stoi(s);
    } catch (...) {
        return std::nullopt;
    }
}

auto result = parse("123");
if (result) {
    std::cout << *result;
}

    内存布局

    // 简化实现
template<typename T>
class optional {
    alignas(T) unsigned char storage[sizeof(T)];
    bool has_value_;
};

// 大小示例
sizeof(std::optional<int>)    == 8   // 4 + padding + 1
sizeof(std::optional<char>)   == 2   // 1 + 1

    2.3 std::any - 类型擦除容器

    基本用法

    #include <any>

// 创建
std::any a1;                     // 空
std::any a2 = 42;                // int
std::any a3 = std::string("hello");  // string

// 赋值（任意类型）
a1 = 3.14;
a1 = std::vector<int>{1, 2, 3};

// 检查
if (a2.has_value()) {
    std::cout << a2.type().name();  // RTTI类型信息
}

// 访问
try {
    int i = std::any_cast<int>(a2);  // OK
    std::string s = std::any_cast<std::string>(a2);  // 抛异常
} catch (const std::bad_any_cast& e) {
    std::cout << "Wrong type\n";
}

// 指针访问（不抛异常）
if (auto p = std::any_cast<int>(&a2)) {
    std::cout << *p;
}

    小对象优化（SBO）

    // 简化实现概念
class any {
    union {
        void* heap_ptr;                // 大对象：堆分配
        alignas(max_align_t) 
        char stack_storage[16];  // 小对象：栈存储
    };
    const std::type_info* type;
};
    对象大小 存储位置 性能
    ≤ 16 bytes 栈上（SBO） 快（无堆分配）
    > 16 bytes 堆上 慢（需new/delete）

    2.4 std::function - 函数对象包装器

    基本用法

    #include <functional>

// 函数指针
int add(int a, int b) { return a + b; }
std::function<int(int, int)> f1 = add;

// Lambda
std::function<int(int, int)> f2 = [](int a, int b) { 
    return a + b; 
};

// 成员函数
struct Calculator {
    int add(int a, int b) { return a + b; }
};
Calculator calc;
std::function<int(Calculator&, int, int)> f3 = &Calculator::add;
f3(calc, 1, 2);

// 使用std::bind
auto f4 = std::bind(&Calculator::add, &calc, std::placeholders::_1, std::placeholders::_2);

// 调用
int result = f1(1, 2);

// 检查
if (f1) {  // 或 f1.operator bool()
    std::cout << "Callable\n";
}

    类型擦除实现

    // 简化实现概念
template<typename Signature>
class function;

template<typename R, typename... Args>
class function<R(Args...)> {
    struct callable_base {
        virtual R invoke(Args...) = 0;
        virtual ~callable_base() = default;
    };
    
    template<typename F>
    struct callable : callable_base {
        F func;
        R invoke(Args... args) override {
            return func(args...);
        }
    };
    
    callable_base* ptr;  // 多态指针
};
    特性 说明
    类型擦除 通过虚函数实现
    性能开销 虚函数调用 + 可能堆分配
    灵活性 可存储任何可调用对象
    SBO优化 小对象栈上存储

    🛠️ 第三层：应用场景

    3.1 std::variant 应用场景

    场景1：表达式树（AST）

    struct Number { int value; };
struct BinaryOp { 
    char op; 
    std::unique_ptr<Expr> left, right; 
};

using Expr = std::variant<Number, BinaryOp>;

int eval(const Expr& e) {
    return std::visit(overloaded{
        [](const Number& n) { return n.value; },
        [](const BinaryOp& op) {
            int l = eval(*op.left);
            int r = eval(*op.right);
            return op.op == '+' ? l + r : l - r;
        }
    }, e);
}

    场景2：状态机

    struct Idle {};
struct Running { int progress; };
struct Completed { int result; };
struct Error { std::string message; };

using State = std::variant<Idle, Running, Completed, Error>;

class Task {
    State state;
    
public:
    void start() {
        std::visit(overloaded{
            [](Idle) { /* 开始执行 */ },
            [](auto) { throw std::runtime_error("Invalid state"); }
        }, state);
        state = Running{0};
    }
};

    3.2 std::optional 应用场景

    场景1：查找操作

    std::optional<std::string> find_user(int id) {
    if (/* 找到 */) {
        return "John";
    }
    return std::nullopt;
}

auto user = find_user(42);
std::cout << user.value_or("Unknown") << "\n";

    场景2：配置参数

    struct Config {
    std::optional<int> port;
    std::optional<std::string> host;
    
    int get_port() const {
        return port.value_or(8080);  // 默认端口
    }
};

    场景3：错误处理

    std::optional<int> safe_divide(int a, int b) {
    if (b == 0) return std::nullopt;
    return a / b;
}

auto result = safe_divide(10, 0);
if (!result) {
    std::cout << "Division by zero\n";
}

    3.3 std::any 应用场景

    场景1：配置存储

    std::map<std::string, std::any> config;
config["port"] = 8080;
config["host"] = std::string("localhost");
config["enabled"] = true;

int port = std::any_cast<int>(config["port"]);

    场景2：消息传递

    struct Message {
    std::string type;
    std::any data;
};

void process(const Message& msg) {
    if (msg.type == "int") {
        int value = std::any_cast<int>(msg.data);
    } else if (msg.type == "string") {
        auto value = std::any_cast<std::string>(msg.data);
    }
}

    3.4 std::function 应用场景

    场景1：回调函数

    class Button {
    std::function<void()> onClick;
    
public:
    void setOnClick(std::function<void()> callback) {
        onClick = callback;
    }
    
    void click() {
        if (onClick) onClick();
    }
};

Button btn;
btn.setOnClick([]{ std::cout << "Clicked!\n"; });
btn.click();

    场景2：策略模式

    class Sorter {
    std::function<bool(int, int)> comparator;
    
public:
    void sort(std::vector<int>& v) {
        std::sort(v.begin(), v.end(), comparator);
    }
    
    void setComparator(std::function<bool(int, int)> cmp) {
        comparator = cmp;
    }
};

Sorter s;
s.setComparator([](int a, int b) { return a > b; });

    🚀 第四层：高级技巧与最佳实践

    4.1 性能优化策略

    variant vs 继承多态

    特性 std::variant 继承多态
    内存分配 ✅ 栈上，无堆分配 ❌ 堆分配（new）
    调用开销 ✅ 可内联 ❌ 虚函数调用
    缓存友好 ✅ 数据连续 ❌ 指针跳转
    类型扩展 ❌ 需修改定义 ✅ 易扩展 
    // ✅ 高性能：variant
std::vector<std::variant<Circle, Square>> shapes;

// ❌ 较慢：继承多态
std::vector<std::unique_ptr<Shape>> shapes;

    optional vs 指针

    // ✅ 推荐：语义清晰
std::optional<int> find(int key);

// ❌ 不推荐：容易误用
int* find(int key);  // 谁负责释放？

    function vs Lambda直接传递

    // ❌ 性能开销大
void process(std::function<void(int)> f) {
    f(42);  // 虚函数调用
}

// ✅ 零开销（模板）
template<typename F>
void process(F f) {
    f(42);  // 可内联
}

    4.2 与模板结合使用

    泛型optional处理

    template<typename T, typename F>
auto map(std::optional<T> opt, F func) 
    -> std::optional<decltype(func(*opt))> 
{
    if (opt) {
        return func(*opt);
    }
    return std::nullopt;
}

auto result = map(
    parse_int("42"),
    [](int x) { return x * 2; }
);

    variant访问器模板

    template<typename... Ts>
auto to_string(std::variant<Ts...> v) {
    return std::visit([](auto&& arg) -> std::string {
        using T = std::decay_t<decltype(arg)>;
        if constexpr (std::is_same_v<T, std::string>) {
            return arg;
        } else {
            return std::to_string(arg);
        }
    }, v);
}

    4.3 错误处理最佳实践

    variant + monostate 表示空状态

    using Result = std::variant<std::monostate, int, std::string>;

Result compute() {
    if (/* error */) return std::monostate{};
    if (/* int result */) return 42;
    return std::string("result");
}

    optional链式调用

    // C++23 std::optional::and_then
auto result = get_user(id)
    .and_then([](auto u) { return get_address(u); })
    .and_then([](auto a) { return get_city(a); })
    .value_or("Unknown");

    4.4 现代C++惯用法

    使用结构化绑定

    std::map<std::string, std::optional<int>> scores;

for (auto&& [name, score] : scores) {
    if (score) {
        std::cout << name << ": " << *score << "\n";
    }
}

    RAII + std::function

    class ScopeGuard {
    std::function<void()> cleanup;
public:
    ScopeGuard(std::function<void()> f) : cleanup(f) {}
    ~ScopeGuard() { if (cleanup) cleanup(); }
};

void process() {
    ScopeGuard guard([]{ std::cout << "Cleanup\n"; });
    // ... 操作 ...
}  // 自动调用cleanup
    💡 使用建议
    • ✅ variant：类型固定、性能敏感场景
    • ✅ optional：替代指针表示"无值"
    • ✅ any：类型完全动态的场景（谨慎使用）
    • ✅ function：回调、策略模式（注意性能）
    • ⚠️ 优先使用模板而非std::function（性能）
    • ⚠️ 避免过度使用std::any（类型安全）
