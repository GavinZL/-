    🎯 第一层：概念与背景

    1.1 void_t的定义

    📌 完整定义 
    namespace std {
    template<typename... Ts>
    using void_t = void;
}

    无论传入什么类型，始终返回void。

    1.2 为什么需要void_t

    传统方法（C++11）

    template<typename T>
class has_begin_old {
    template<typename U>
    static auto test(int) 
        -> decltype(std::declval<U>().begin(), std::true_type{});
    template<typename>
    static std::false_type test(...);
public:
    using type = decltype(test<T>(0));
};

    void_t方法（C++17）

    template<typename T, typename = void>
struct has_begin : std::false_type {};

template<typename T>
struct has_begin<T, std::void_t<decltype(std::declval<T>().begin())>>
    : std::true_type {};
    特性 传统SFINAE void_t
    代码行数 9行 5行
    可读性 ❌ ✅
    维护性 ❌ ✅

    ⚙️ 第二层：核心机制

    2.1 工作原理

    template<typename T, typename = void>
struct has_value_type : std::false_type {};

template<typename T>
struct has_value_type<T, std::void_t<typename T::value_type>>
    : std::true_type {};

    编译器处理流程

    ✅ vector<int>
    T::value_type → int
    void_t<int> → void
    匹配特化版本
    结果：true
    ❌ int
    T::value_type → 错误
    SFINAE触发
    使用主模板
    结果：false

    🛠️ 第三层：应用场景

    3.1 条件性优化

    template<typename T, typename = void>
struct has_reserve : std::false_type {};

template<typename T>
struct has_reserve<T, std::void_t<
    decltype(std::declval<T>().reserve(std::declval<size_t>()))
>> : std::true_type {};

template<typename C>
void prepare(C& c, size_t n) {
    if constexpr (has_reserve<C>::value) {
        c.reserve(n);  // vector有
    }
    // list无，跳过
}

    3.2 Detection Idiom

    template<typename Default, typename Void,
         template<typename...> class Op, typename... Args>
struct detector {
    using value_t = std::false_type;
    using type = Default;
};

template<typename Default, template<typename...> class Op, 
         typename... Args>
struct detector<Default, std::void_t<Op<Args...>>, Op, Args...> {
    using value_t = std::true_type;
    using type = Op<Args...>;
};

    3.3 通用打印

    template<typename T, typename = void>
struct is_printable : std::false_type {};

template<typename T>
struct is_printable<T, std::void_t<
    decltype(std::cout << std::declval<T>())
>> : std::true_type {};

template<typename T>
void print(const T& v) {
    if constexpr (is_printable<T>::value) {
        std::cout << v << "\n";
    } else {
        std::cout << "[not printable]\n";
    }
}

    4.3 常见陷阱

    陷阱1：缺少主模板

    // ❌ 错误
template<typename T>
struct has_begin<T, std::void_t<...>> : std::true_type {};

// ✅ 正确
template<typename T, typename = void>
struct has_begin : std::false_type {};

template<typename T>
struct has_begin<T, std::void_t<...>> : std::true_type {};

    陷阱2：C++14兼容性

    // C++14需自己实现
#if __cplusplus < 201703L
template<typename... Ts>
using void_t = void;
#endif
