🧠 4.1 模板元编程 (TMP)

编译期计算

// 编译期阶乘计算
template<int N>
struct Factorial {
    static constexpr int value = N * Factorial<N-1>::value;
};

template<>
struct Factorial<0> {
    static constexpr int value = 1;
};

// 使用
constexpr int result = Factorial<5>::value;  // 编译期计算：120

类型计算

// 类型萃取：移除const
template<typename T>
struct remove_const {
    using type = T;
};

template<typename T>
struct remove_const<const T> {
    using type = T;
};

// 使用
using Type1 = remove_const<const int>::type;  // int
using Type2 = remove_const<int>::type;        // int
💡 现代替代方案 C++11后，优先使用constexpr函数替代模板元编程，代码更清晰、编译更快。

🎭 4.2 SFINAE技术

基本原理

SFINAE（Substitution Failure Is Not An Error）：替换失败不是错误。当模板参数替换失败时，编译器不会报错，而是从候选集中移除该模板。

// 使用enable_if进行SFINAE
template<typename T>
typename std::enable_if<std::is_integral<T>::value, T>::type
process(T value) {
    return value * 2;  // 仅对整型有效
}

template<typename T>
typename std::enable_if<std::is_floating_point<T>::value, T>::type
process(T value) {
    return value / 2.0;  // 仅对浮点型有效
}

检测成员函数是否存在

// 使用void_t检测成员
template<typename, typename = void>
struct has_size : std::false_type {};

template<typename T>
struct has_size<T, std::void_t<decltype(std::declval<T>().size())>>
    : std::true_type {};

// 使用
static_assert(has_size<std::vector<int>>::value);  // true
static_assert(!has_size<int>::value);           // true

📦 4.3 变参模板

参数包展开

// 递归展开
void print() {}  // 终止函数

template<typename T, typename... Args>
void print(T first, Args... rest) {
    std::cout << first << " ";
    print(rest...);  // 递归调用
}

print(1, 2.5, "hello");  // 输出：1 2.5 hello

C++17折叠表达式

// 一元右折叠：求和
template<typename... Args>
auto sum(Args... args) {
    return (args + ...);
}

// 一元左折叠：打印
template<typename... Args>
void print(Args... args) {
    (std::cout << ... << args) << "\\n";
}

sum(1, 2, 3, 4);  // 10
print(1, " ", 2.5, " ", "hello");  // 1 2.5 hello

索引序列技巧

// 使用索引序列展开元组
template<typename Tuple, size_t... Is>
void print_tuple_impl(Tuple&& t, std::index_sequence<Is...>) {
    ((std::cout << std::get<Is>(t) << " "), ...);
}

template<typename... Args>
void print_tuple(const std::tuple<Args...>& t) {
    print_tuple_impl(t, std::index_sequence_for<Args...>{});
}

✨ 4.4 C++20 Concepts

定义Concept

// 定义数值概念
template<typename T>
concept Numeric = std::is_arithmetic_v<T>;

// 定义容器概念
template<typename T>
concept Container = requires(T c) {
    { c.begin() } -> std::same_as<typename T::iterator>;
    { c.end() } -> std::same_as<typename T::iterator>;
    { c.size() } -> std::convertible_to<size_t>;
};

使用Concept约束模板

// 方式1：requires子句
template<typename T>
    requires Numeric<T>
T add(T a, T b) {
    return a + b;
}

// 方式2：简化语法
template<Numeric T>
T multiply(T a, T b) {
    return a * b;
}

// 方式3：简写函数模板
auto divide(Numeric auto a, Numeric auto b) {
    return a / b;
}
⚠️ Concepts的优势 相比SFINAE，Concepts提供：
1. 更清晰的约束表达
2. 更友好的错误信息（从数百行降至几行）
3. 更精确的重载决议
4. 编译速度提升

⚡ 4.5 完美转发

万能引用与引用折叠

// 万能引用：T&&
template<typename T>
void wrapper(T&& arg) {
    process(std::forward<T>(arg));  // 完美转发
}

int x = 10;
wrapper(x);        // T推导为int&，arg是左值引用
wrapper(10);       // T推导为int，arg绑定到右值

引用折叠规则

原始类型& && &&&& &&& &&
折叠结果&&&&&

实现make_unique

template<typename T, typename... Args>
std::unique_ptr<T> make_unique(Args&&... args) {
    return std::unique_ptr<T>(
        new T(std::forward<Args>(args)...)
    );
}

// 使用
auto ptr = make_unique<std::string>("Hello");  // 右值
std::string s = "World";
auto ptr2 = make_unique<std::string>(s);      // 左值
