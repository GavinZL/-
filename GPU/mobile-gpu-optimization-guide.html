2. Draw Call 与渲染批次优化

2.1 为什么 Draw Call 是移动端首要瓶颈

在移动端，单次 glDraw* 的 CPU 成本通常在 10–50µs，包括状态验证、命令打包和驱动调度。 对于中低端设备，50–100 个未优化 Draw Call 就足以占满一半以上 CPU 帧时间。因此，从引擎层面控制 DC 数量和状态切换，是 GPU 优化的第一优先级。

典型目标（60fps）
  • 高端机（8 Gen2 级）：≤ 150 DC/帧。
  • 中端机（778G 级）：≤ 80 DC/帧。
  • 低端机（660 级）：≤ 50 DC/帧。
  • UI / 2D 场景：尽量压到 20–50 DC/帧。

2.2 Bad Case：每对象一条 Draw Call + 频繁状态切换

// ❌ 错误：每个精灵单独 Draw，频繁绑定纹理/Shader
void RenderNaive(const std::vector<Sprite>& sprites) {
    for (const auto& s : sprites) {
        glBindTexture(GL_TEXTURE_2D, s.texture);     // 频繁切换
        glUseProgram(s.program);                    // 频繁切换

        glBindBuffer(GL_ARRAY_BUFFER, vbo);
        glBufferData(GL_ARRAY_BUFFER, sizeof(Vertex) * 4,
                     s.vertices, GL_DYNAMIC_DRAW);  // 每精灵一次上传

        glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_SHORT, nullptr);
    }
}
// 100 个精灵 ≈ 100 次 Draw + 100 次纹理绑定 + 100 次 glBufferData
// 在低端机上可能消耗 5–8ms CPU 时间，仅用于 2D 精灵渲染。

2.3 Good Case：按状态排序 + 状态缓存

// ✅ 状态缓存：避免冗余 glBindTexture / glUseProgram 调用
struct StateCache {
    GLuint curTex = 0;
    GLuint curProg = 0;
    void bindTexture(GLuint tex) {
        if (tex != curTex) { glBindTexture(GL_TEXTURE_2D, tex); curTex = tex; }
    }
    void useProgram(GLuint p) {
        if (p != curProg) { glUseProgram(p); curProg = p; }
    }
};

struct DrawCmd {
    uint32_t sortKey;   // Shader(8) | Texture(16) | Depth(8)
    Sprite*  sprite;
};

void RenderSorted(std::vector<DrawCmd>& cmds) {
    for (auto& c : cmds) {
        auto& s = *c.sprite;
        uint32_t shader  = (s.program & 0xFF)   << 24;
        uint32_t texture = (s.texture & 0xFFFF) << 8;
        uint32_t depth   = (uint32_t)(s.depth * 255.f) & 0xFF;
        c.sortKey = shader | texture | depth;
    }
    std::sort(cmds.begin(), cmds.end(),
              [](const DrawCmd& a, const DrawCmd& b){ return a.sortKey < b.sortKey; });

    StateCache state;
    for (auto& c : cmds) {
        auto& s = *c.sprite;
        state.useProgram(s.program);
        state.bindTexture(s.texture);
        DrawSprite(s);
    }
}
// 改造后：纹理/Shader 切换次数接近“种类数”，而非“物体数”。

2.4 Good Case：动态批处理（Dynamic Batching）

// ✅ 动态批处理：将同纹理精灵装入一个大 VBO，一次 Draw 画完
class DynamicBatcher {
public:
    static constexpr size_t MAX_SPRITES = 1000;
    Vertex   verts[MAX_SPRITES * 4];
    uint16_t idx  [MAX_SPRITES * 6];
    uint32_t count = 0;
    GLuint vbo = 0, ibo = 0;
    GLuint curTex = 0, curProg = 0;

    void init() {
        glGenBuffers(1, &vbo);
        glGenBuffers(1, &ibo);
    }

    void begin(GLuint tex, GLuint prog) {
        curTex = tex; curProg = prog; count = 0;
    }

    void addSprite(const Sprite& s) {
        if (s.texture != curTex || count >= MAX_SPRITES) {
            flush();
            begin(s.texture, s.program);
        }
        uint32_t baseV = count * 4;
        memcpy(&verts[baseV], s.vertices, sizeof(Vertex) * 4);

        uint32_t baseI = count * 6;
        idx[baseI+0] = baseV+0; idx[baseI+1] = baseV+1; idx[baseI+2] = baseV+2;
        idx[baseI+3] = baseV+2; idx[baseI+4] = baseV+3; idx[baseI+5] = baseV+0;
        ++count;
    }

    void flush() {
        if (!count) return;
        glUseProgram(curProg);
        glBindTexture(GL_TEXTURE_2D, curTex);

        glBindBuffer(GL_ARRAY_BUFFER, vbo);
        glBufferData(GL_ARRAY_BUFFER, count*4*sizeof(Vertex),
                     verts, GL_DYNAMIC_DRAW);
        glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, ibo);
        glBufferData(GL_ELEMENT_ARRAY_BUFFER, count*6*sizeof(uint16_t),
                     idx, GL_DYNAMIC_DRAW);

        setupVertexAttributes();
        glDrawElements(GL_TRIANGLES, count*6, GL_UNSIGNED_SHORT, nullptr);
        count = 0;
    }
};
// 实测：100 精灵从 100 个 Draw Call 压缩到 ~5 个，CPU 开销可下降 80–90%。

3. 着色器优化（精度 / 运算 / 控制流 / 采样）

3.1 精度控制：lowp / mediump / highp

移动 GPU 在 FP16 通路上极具优势：lowp/mediump 通常是 highp 的两倍以上吞吐。 因此，片元着色器必须遵循“能低不用高”的精度策略。

经验规则
  • 颜色（color/alpha）：lowp。
  • 纹理坐标、法线、一般向量：mediump。
  • 位置/深度/大范围距离：highp，且尽量在 VS 计算。

Bad Case：全局 highp 滥用

// ❌ 片元着色器中过度使用 highp，浪费寄存器与带宽
precision highp float;

varying highp vec2 v_texCoord;
varying highp vec4 v_color;
uniform highp sampler2D u_tex;

void main() {
    highp vec4 t = texture2D(u_tex, v_texCoord);
    gl_FragColor = t * v_color;  // 所有运算都走 FP32 通路
}

Good Case：最小精度原则

// ✅ 精度划分合理：默认 mediump，颜色 lowp，采样结果 lowp
precision mediump float;

varying mediump vec2 v_texCoord;
varying lowp    vec4 v_color;
uniform sampler2D u_tex;

void main() {
    lowp vec4 texColor = texture2D(u_tex, v_texCoord);
    gl_FragColor = texColor * v_color;
}
// 实测：在 Mali 上 cycles/pixel 从 ~1.3 降到 ~0.6，功耗降低 ~30–40%。

3.2 算术运算：避免昂贵指令

除法、sqrt、pow、三角函数都是移动 GPU 上的重灾区，需用乘法、inversesqrt、查找表等替代。

Bad Case：多次除法和 sqrt

// ❌ 每像素多次除法和 sqrt
float len = sqrt(dot(n, n));    // sqrt
n = n / len;                    // 除法
float k = pow(brightness, 2.2); // pow

Good Case：inversesqrt + 乘法近似

// ✅ 使用 inversesqrt 和近似幂函数
float invLen = inversesqrt(dot(n, n));
n = n * invLen;  // 乘法代替除法

// Gamma 校正用查找表，或拆成简单幂次
// 例如 pow(x, 2.2) ≈ x * x * sqrt(x) 或查 LUT 纹理

3.3 控制流与分支发散

大量 if/else 会使同一 wave 内的像素走不同分支，导致 SIMT 串行化。 引擎侧应优先使用 Shader 变体（编译期分支） 和 mix/step 等替代简单运行期分支。

Bad Case：运行期多分支材质

// ❌ 在片元着色器中根据 materialType 切换复杂路径
if (materialType == 0) {
    color = CalcPBR();
} else if (materialType == 1) {
    color = CalcPhong();
} else {
    color = CalcUnlit();
}

Good Case：编译期变体 + 简单控制流

// ✅ 通过宏生成多个 Shader 变体
#ifdef USE_PBR
    vec3 color = CalcPBR();
#elif defined(USE_PHONG)
    vec3 color = CalcPhong();
#else
    vec3 color = CalcUnlit();
#endif

4. 纹理与采样优化（压缩 / 图集 / Mipmap）

4.1 压缩格式：ETC2 / ASTC / PVRTC

在统一内存架构上，纹理读写直接抢系统带宽。使用硬件压缩格式可以将同等画质下的纹理带宽压缩到原来的 1/4～1/16。 典型策略：Android 以 ETC2/ASTC 为主，iOS 以 ASTC（新）/PVRTC（旧）为主。

4.2 纹理图集（Texture Atlas）

UI 与 2D 精灵强烈建议使用纹理图集，配合批处理可将几十上百张小纹理合并为少量 Draw Call。

Bad Case：每个 UI 元素单独纹理

// ❌ 50 个 UI 元素，50 张纹理，50 次 Draw Call
for (auto& e : uiElements) {
    glBindTexture(GL_TEXTURE_2D, e.texture);
    DrawQuad(e);
}

Good Case：纹理图集 + 批处理

// ✅ 构建图集后，所有 UI 使用同一纹理，只调整 UV
struct Region { float u0, v0, u1, v1; };
Region atlasRegion(int imageId); // 打包阶段生成

void RenderUIOptimized(const std::vector<UIElement>& elements) {
    glBindTexture(GL_TEXTURE_2D, g_uiAtlasTex);
    DynamicBatcher batch;
    batch.begin(g_uiAtlasTex, g_uiShader);

    for (auto& e : elements) {
        Region uv = atlasRegion(e.imageId);
        Sprite s = buildSpriteFromUI(e, uv); // 顶点写入图集 UV
        batch.addSprite(s);
    }
    batch.flush();
}
// Draw Call 从 50 次收敛到 1–3 次，CPU/GPU 压力大幅降低。

5. 内存与带宽：VBO/IBO、多缓冲与数据同步

5.1 VBO/IBO 策略与多缓冲

每帧频繁创建/销毁 VBO 或对正在使用的缓冲调用 glBufferData，都会导致驱动插入同步点或产生严重碎片。 引擎应：

  • 为动态数据使用持久化 VBO，只在帧间 glBufferSubData 更新内容。
  • 对高频更新数据采用双/三缓冲 VBO，避免 CPU 等待 GPU。
  • 为 VBO/FBO/纹理等建立对象池。

Bad Case：每帧重建 VBO

// ❌ 每帧删除+创建 VBO，使用 STATIC_DRAW
void RenderUI_Bad(const UI& ui) {
    if (vbo) glDeleteBuffers(1, &vbo);
    glGenBuffers(1, &vbo);
    glBindBuffer(GL_ARRAY_BUFFER, vbo);

    std::vector<Vertex> vertices;
    ui.buildVertices(vertices);

    glBufferData(GL_ARRAY_BUFFER,
                 vertices.size()*sizeof(Vertex),
                 vertices.data(), GL_STATIC_DRAW);
    DrawVBO();
}

Good Case：持久化 VBO + SubData

// ✅ 初始化时一次性分配，帧间仅更新内容
class PersistentVBO {
public:
    void init(size_t maxBytes) {
        glGenBuffers(1, &vbo);
        glBindBuffer(GL_ARRAY_BUFFER, vbo);
        glBufferData(GL_ARRAY_BUFFER, maxBytes, nullptr, GL_DYNAMIC_DRAW);
    }

    void updateAndDraw(const std::vector<Vertex>& verts) {
        glBindBuffer(GL_ARRAY_BUFFER, vbo);
        glBufferSubData(GL_ARRAY_BUFFER, 0,
                        verts.size()*sizeof(Vertex), verts.data());
        setupAttribs();
        glDrawArrays(GL_TRIANGLES, 0, (GLsizei)verts.size());
    }
private:
    GLuint vbo = 0;
};

5.2 PBO：异步纹理上传

对于 1080p 视频/大纹理，使用 PBO 可将上传延迟从十几毫秒降到几毫秒，并大幅减少 CPU 阻塞时间。

Bad Case：同步 TexSubImage 上传视频帧

// ❌ 每帧直接上传 1920x1080 RGBA 像素
void UploadFrame_Bad(GLuint tex, const void* pixels, int w, int h) {
    glBindTexture(GL_TEXTURE_2D, tex);
    glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, w, h,
                    GL_RGBA, GL_UNSIGNED_BYTE, pixels);
    // CPU 可能在这一步等待 GPU 使用完上一帧数据
}

Good Case：PBO 双缓冲上传

// ✅ 通过 PBO 双缓冲实现异步上传
class AsyncUploader {
    GLuint pbos[2]; GLuint tex; int frame = 0; int w, h;
public:
    void init(int width, int height) {
        w = width; h = height;
        glGenBuffers(2, pbos);
        for (int i = 0; i < 2; ++i) {
            glBindBuffer(GL_PIXEL_UNPACK_BUFFER, pbos[i]);
            glBufferData(GL_PIXEL_UNPACK_BUFFER, w*h*4, nullptr, GL_STREAM_DRAW);
        }
        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, 0);

        glGenTextures(1, &tex);
        glBindTexture(GL_TEXTURE_2D, tex);
        glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, w, h, 0,
                     GL_RGBA, GL_UNSIGNED_BYTE, nullptr);
    }

    void upload(const void* pixels) {
        int cur = frame % 2;
        int prev = (frame + 1) % 2;

        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, pbos[cur]);
        glBufferData(GL_PIXEL_UNPACK_BUFFER, w*h*4, pixels, GL_STREAM_DRAW);

        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, pbos[prev]);
        glBindTexture(GL_TEXTURE_2D, tex);
        glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, w, h,
                        GL_RGBA, GL_UNSIGNED_BYTE, nullptr);
        glBindBuffer(GL_PIXEL_UNPACK_BUFFER, 0);
        ++frame;
    }
};

6. 移动 GPU 架构差异（Mali / Adreno / PowerVR）

三大主流移动 GPU（Mali、Adreno、PowerVR）在宏观上都是 TBDR/Tile-based 架构， 但在精度、分支处理、工具链上各有偏好，引擎在做跨平台优化时需考虑这些差异。

架构特点优化重点
Mali FP16 性能极好；Mali Offline Compiler 支持详尽 cycles/pixel 分析。 充分使用 mediump/lowp；避免 overdraw；利用官方工具查看指令与瓶颈。
Adreno instancing/批处理表现好；Snapdragon Profiler 提供丰富性能计数器。 积极使用 instancing；关注带宽与 alpha-blend；利用 GPU Counters 校正假设。
PowerVR 典型 TBDR；对「中途读取当前 RT」极敏感。 避免在同一 RenderPass 内采样当前 RT；减少 FBO 切换；用 discard/invalidate 减少 resolve。

7. 性能目标与调试流程

7.1 推荐性能目标

场景类型帧率目标Draw Call备注
UI / 2D60fps< 50（理想 20–30）大量图集 + 批处理
轻度 3D60fps< 150适度 LOD 与 instancing
重度 3D30–60fps< 200重剔除 + LOD + GPU 驱动渲染

7.2 CPU/GPU 分离诊断流程

  1. 用 timer query 或平台 profiler，获取GPU 帧时间与CPU 帧时间。
  2. 通过“分辨率测试”：降分辨率，如果帧率大幅提升 → Fragment/带宽瓶颈。
  3. 通过“着色器测试”：将主要 FS 替换为简单颜色输出，观察变化 → Shader/纹理瓶颈。
  4. 结合 Draw Call 数、纹理尺寸、overdraw 视图，定位问题模块。

8. 引擎级 GPU 优化 Checklist（可嵌入评审流程）

本节给出一套按模块划分的引擎级 GPU 优化 Checklist，可直接作为代码评审 / 性能门禁表使用。 建议在功能评审与版本打包前，按照 Checklist 逐项勾检。

编号类别评审项检查要点 / 度量方式通过标准
DC-01 Draw Call 渲染队列排序 是否按 (Shader, 纹理, RenderState, Depth) 组装 sort key 并排序执行？ 主渲染路径中，所有 Mesh/Sprite 通过统一 RenderQueue 排序后再提交。
DC-02 Draw Call 状态缓存 是否有集中管理的 StateCache，避免冗余 glBindTexture/glUseProgram 调用？ 同一帧内，同一资源不会被重复绑定超过 1 次（除上下文丢失）。
DC-03 Draw Call 批处理 2D/UI/粒子是否使用动态批处理或 instancing，将同材质对象合并 Draw？ 常驻场景中，2D/UI Draw Call ≤ 50，粒子系统使用 instancing/批处理。
SH-01 Shader 精度规范 Fragment 是否默认 mediump，颜色统一使用 lowp，仅必要位置/深度使用 highp？ 离线编译中 FP16 使用率 > 70%；无全局 precision highp float;。
SH-02 Shader 昂贵函数 是否避免在 FS 中出现多次 pow/exp/log/sin/cos，必要时是否使用 LUT/近似？ 主要 FS 中高成本函数出现次数 ≤ 1，且对性能有测量与说明。
SH-03 Shader 控制流 是否使用 Shader 变体取代运行期材质大分支？ FS 中不出现材质类型大分支；多材质通过宏/编译变体实现。
TX-01 纹理 压缩与 mipmap 所有大纹理是否使用 ETC2/ASTC/PVRTC 等硬件压缩并开启 mipmap？ 显存中 ≥ 90% 像素纹理为压缩格式；除 UI 小图外全部启用 mipmap。
TX-02 纹理 图集与数组 UI/精灵是否打成图集或 Texture Array，以减少纹理切换？ 主 UI 体系不使用单图散落纹理；同类图标/精灵打入少数图集。
MB-01 内存 VBO 生命周期 动态几何是否使用持久化 VBO + glBufferSubData 或多缓冲？ 渲染帧循环中禁止创建/删除 VBO、禁止频繁 glBufferData 重分配。
MB-02 带宽 RT 使用 是否避免在同一 RenderPass 内读取当前 RT（打断 TBDR）？是否在不再需要的 depth/stencil 上调用 invalidate？ 无「先写 RT 再立即采样 RT」的模式；结束 FBO 时对无用附件进行 discard/invalidate。
MB-03 带宽 视频/摄像头上传 是否对 1080p 级视频/摄像头流使用 PBO 或 HardwareBuffer/CVPixelBuffer 零拷贝？ 1080p 视频上传 GPU 时间 < 3ms；摄像头预览无明显卡顿与大幅 CPU 峰值。
PF-01 流程 性能基线 是否为关键场景配置固定 test case，记录 CPU/GPU 帧时间、DC 数、三角形数、带宽指标？ 每次版本迭代均对关键场景做回归；不允许指标无原因回退。
PF-02 流程 平台工具集成 是否在 CI/日常开发中使用 RenderDoc + Mali/Adreno/Xcode 工具抽查 GPU 状态？ 每个大版本至少执行一次全链路 GPU 分析并形成结论文档。

将上述 Checklist 嵌入引擎/项目的渲染模块评审模板和性能门禁脚本中，可以显著降低“性能回退”“上线后过热掉帧”等风险。
