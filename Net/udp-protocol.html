    1.1 UDP设计哲学

    📌 历史背景
    • 1980年：RFC 768发布（仅3页）
    • 设计者：David P. Reed
    • 目标：最简单的传输层协议
    • 核心：端到端原则

    1.2 UDP vs TCP

    特性 UDP TCP
    连接 无连接 面向连接
    可靠性 不可靠 可靠
    头部 8字节 20-60字节
    速度 快 较慢
    消息边界 保留 字节流

    2.2 Socket编程

    2.2.1 UDP服务器实现

    /* UDP Echo服务器 */
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <string.h>
#include <stdio.h>

int main() {
    // 1. 创建UDP套接字
    int sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    if (sockfd < 0) {
        perror("socket");
        return 1;
    }
    
    // 2. 绑定地址和端口
    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(8888);
    server_addr.sin_addr.s_addr = INADDR_ANY;  // 监听所有接口
    
    if (bind(sockfd, (struct sockaddr*)&server_addr, 
             sizeof(server_addr)) < 0) {
        perror("bind");
        return 1;
    }
    
    printf("UDP Server listening on port 8888...\n");
    
    // 3. 接收和发送数据
    char buffer[1024];
    struct sockaddr_in client_addr;
    socklen_t client_len;
    
    while(1) {
        client_len = sizeof(client_addr);
        
        // recvfrom: 接收数据并获取发送者地址
        int n = recvfrom(sockfd, buffer, sizeof(buffer) - 1, 0,
                          (struct sockaddr*)&client_addr, &client_len);
        
        if (n > 0) {
            buffer[n] = '\0';
            printf("Received from %s:%d - %s\n",
                   inet_ntoa(client_addr.sin_addr),
                   ntohs(client_addr.sin_port),
                   buffer);
            
            // sendto: 发送数据到指定地址
            sendto(sockfd, buffer, n, 0,
                   (struct sockaddr*)&client_addr, client_len);
        }
    }
    
    return 0;
}

    2.2.2 UDP客户端实现

    /* UDP客户端 */
int main() {
    // 1. 创建UDP套接字
    int sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    
    // 2. 设置服务器地址
    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(8888);
    inet_pton(AF_INET, "127.0.0.1", &server_addr.sin_addr);
    
    // 3. 发送数据（无需connect）
    const char* msg = "Hello UDP!";
    sendto(sockfd, msg, strlen(msg), 0,
           (struct sockaddr*)&server_addr, sizeof(server_addr));
    
    // 4. 接收响应
    char buffer[1024];
    struct sockaddr_in from_addr;
    socklen_t from_len = sizeof(from_addr);
    
    int n = recvfrom(sockfd, buffer, sizeof(buffer), 0,
                      (struct sockaddr*)&from_addr, &from_len);
    
    buffer[n] = '\0';
    printf("Received: %s\n", buffer);
    
    close(sockfd);
    return 0;
}

    2.3 Linux内核UDP实现

    2.3.1 UDP套接字结构

    /* net/ipv4/udp.c - Linux内核UDP实现 */

struct udp_sock {
    struct inet_sock inet;
    int pending;           // 待处理的数据
    unsigned int corkflag;  // Cork标志
    __u16 encap_type;       // 封装类型
    /* 其他字段 */
};

/* UDP接收路径 */
int udp_rcv(struct sk_buff *skb) {
    // 1. 获取UDP头部
    struct udphdr *uh = udp_hdr(skb);
    
    // 2. 校验和检查
    if (udp_checksum_complete(skb))
        goto csum_error;
    
    // 3. 查找对应的socket
    struct sock *sk = __udp4_lib_lookup(skb);
    
    // 4. 将数据放入socket接收队列
    if (sk)
        return udp_queue_rcv_skb(sk, skb);
    
    // 5. 没有匹配的socket，发送ICMP端口不可达
    icmp_send(skb, ICMP_DEST_UNREACH, ICMP_PORT_UNREACH, 0);
    return 0;
}

/* UDP发送路径 */
int udp_sendmsg(struct sock *sk, struct msghdr *msg, size_t len) {
    // 1. 构造UDP头部
    struct udphdr *uh;
    uh->source = inet_sk(sk)->inet_sport;
    uh->dest = dport;
    uh->len = htons(len);
    
    // 2. 计算校验和
    uh->check = udp_csum(skb);
    
    // 3. 调用IP层发送
    return ip_send_skb(net, skb);
}

    3.1 DNS域名解析

    3.1.1 DNS查询流程

    /* DNS查询使用UDP 53端口 */

Client                    DNS Server
  |                            |
  |  Query: www.google.com?   |
  | -------------------------> |
  |                            | // UDP快速响应
  | Response: 142.250.185.68    |
  | <------------------------- |
  |                            |

/* DNS报文结构 */
struct dns_header {
    uint16_t id;           // 事务ID
    uint16_t flags;        // 标志位
    uint16_t qdcount;      // 问题数
    uint16_t ancount;      // 答案数
    uint16_t nscount;      // 授权数
    uint16_t arcount;      // 额外信息数
};
    💡 为什么DNS使用UDP？
    • 快速响应：无需三次握手，减少延迟
    • 开销小：大量查询不会消耗大量连接资源
    • 简单高效：查询-响应模式完美匹配
    • 限制：响应超过512字节时转用TCP（TC标志）

    3.1.2 DNS实现示例

    /* 简化的DNS查询 */
void dns_query(const char* hostname) {
    int sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    
    // DNS服务器地址 (8.8.8.8:53)
    struct sockaddr_in dns_server;
    dns_server.sin_family = AF_INET;
    dns_server.sin_port = htons(53);
    inet_pton(AF_INET, "8.8.8.8", &dns_server.sin_addr);
    
    // 构造DNS查询报文
    unsigned char query[512];
    int query_len = build_dns_query(query, hostname);
    
    // 发送查询
    sendto(sockfd, query, query_len, 0,
           (struct sockaddr*)&dns_server, sizeof(dns_server));
    
    // 接收响应
    unsigned char response[512];
    int n = recvfrom(sockfd, response, sizeof(response), 0, 
                      NULL, NULL);
    
    // 解析响应
    parse_dns_response(response, n);
    
    close(sockfd);
}

// 服务器: UDP 67

/* DHCP报文结构 */

    uint8_t  op;          // 操作类型 (1=请求, 2=回复)
    uint8_t  htype;       // 硬件类型 (1=Ethernet)
    uint8_t  hlen;        // 硬件地址长度 (6)
    uint8_t  hops;        // 跳数

    uint32_t yiaddr;      // 你的IP

};
    📌 为什么DHCP使用UDP？
    • 广播支持：UDP支持广播，TCP不支持
    • 无需连接：客户端没有IP时无法建立TCP连接
    • 简单高效：配置过程简单，无需复杂协议

    4.2 UDP打洞（NAT穿透）

    4.2.1 NAT类型

    NAT类型 特点 穿透难度
    全锥型 内外网IP:Port一对一映射 易
    限制锥型 仅允许曾通信过的对端 中
    端口限制锥型 限制IP和端口 难
    对称NAT 不同IP使用不同映射 极难

    4.2.2 UDP打洞原理

    /* UDP Hole Punching */

// 场景：Client A和Client B都在NAT后，想要直连

Step 1: A和B分别向Server发送UDP包
        A(private: 192.168.1.10:5000)
        NAT-A分配: A_public(1.2.3.4:10000)
        
        B(private: 192.168.2.20:6000)
        NAT-B分配: B_public(5.6.7.8:20000)

Step 2: Server告诉A，B的公网地址是5.6.7.8:20000
        Server告诉B，A的公网地址是1.2.3.4:10000

Step 3: A向5.6.7.8:20000发送UDP包
        - NAT-A建立映射：192.168.1.10:5000 <-> 1.2.3.4:10000
        - 包到达NAT-B，被丢弃（因为NAT-B不知道这个包）

Step 4: B向1.2.3.4:10000发送UDP包
        - NAT-B建立映射：192.168.2.20:6000 <-> 5.6.7.8:20000
        - 包通过NAT-A，到达A！

Step 5: A再次向5.6.7.8:20000发送
        - 这次可以通过NAT-B，到达B！

Step 6: 洞已打开，A和B可以直接通信

/* 实现代码 */
void udp_hole_punch(const char* peer_ip, int peer_port) {
    int sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    
    // 绑定本地端口
    struct sockaddr_in local_addr;
    local_addr.sin_port = htons(5000);
    bind(sockfd, ...);
    
    // 向对端发送打洞包
    struct sockaddr_in peer_addr;
    inet_pton(AF_INET, peer_ip, &peer_addr.sin_addr);
    peer_addr.sin_port = htons(peer_port);
    
    // 多次发送（增加成功率）
    for(int i = 0; i < 5; i++) {
        sendto(sockfd, "PUNCH", 5, 0,
               (struct sockaddr*)&peer_addr, sizeof(peer_addr));
        usleep(100000);  // 100ms
    }
    
    // 开始正常通信
    // ...
}

    4.3 UDP洪水攻击与防御

    4.3.1 攻击类型

    攻击 原理 防御
    UDP Flood 大量UDP包耗尽带宽 速率限制、防火墙
    放大政击 利用DNS/NTP放大流量 源地址验证
    碎片攻击 发送大量IP碎片 限制碎片重组
    DNS放大 小请求大响应 Response Rate Limiting

    time_t now = time(NULL);
    auto& limit = limits[client_ip];
    
    if (now - limit.last_time > 1) {  // 1秒窗口
        limit.last_time = now;
        limit.count = 1;
        return true;
    }
    
    if (limit.count < 100) {  // 每秒100个包
        limit.count++;
        return true;
    }
    
    return false;  // 超限，拒绝服务
}

    4.4 可靠UDP协议设计

    4.4.1 基本可靠性机制

    /* 自定义可靠UDP协议 */

// 1. 序列号机制
struct rudp_header {
    uint32_t seq;         // 序列号
    uint32_t ack;         // 确认号
    uint16_t flags;       // 标志 (ACK/SYN/FIN)
    uint16_t window;      // 窗口大小
};

// 2. 超时重传
struct pending_packet {
    uint32_t seq;
    char data[1024];
    int len;
    time_t send_time;
    int retry_count;
};

void check_timeout() {
    time_t now = time(NULL);
    
    for(auto& pkt : pending_packets) {
        if (now - pkt.send_time > RTO) {
            if (pkt.retry_count < MAX_RETRY) {
                // 重传
                sendto(sockfd, pkt.data, pkt.len, ...);
                pkt.send_time = now;
                pkt.retry_count++;
            } else {
                // 连接失败
                handle_connection_loss();
            }
        }
    }
}

// 3. ACK机制
void on_receive_packet(uint32_t seq) {
    if (seq == expected_seq) {
        expected_seq++;
        send_ack(seq);
        deliver_to_app();
    } else if (seq > expected_seq) {
        // 乱序，缓存并发送重复ACK
        buffer_packet(seq);
        send_ack(expected_seq - 1);
    }
}

    4.5 UDP广播与组播

    4.5.1 广播

    /* UDP广播 */

// 1. 启用广播
int broadcast = 1;
setsockopt(sockfd, SOL_SOCKET, SO_BROADCAST,
           &broadcast, sizeof(broadcast));

// 2. 发送广播
struct sockaddr_in broadcast_addr;
broadcast_addr.sin_family = AF_INET;
broadcast_addr.sin_port = htons(8888);
inet_pton(AF_INET, "255.255.255.255", &broadcast_addr.sin_addr);

sendto(sockfd, data, len, 0,
       (struct sockaddr*)&broadcast_addr, sizeof(broadcast_addr));

// 3. 接收广播
bind(sockfd, INADDR_ANY, 8888);
recvfrom(sockfd, buffer, len, ...);

    4.5.2 组播

    /* UDP组播 (Multicast) */

// 组播IP范围: 224.0.0.0 - 239.255.255.255

// 1. 发送端
struct sockaddr_in mcast_addr;
mcast_addr.sin_family = AF_INET;
mcast_addr.sin_port = htons(8888);
inet_pton(AF_INET, "239.0.0.1", &mcast_addr.sin_addr);

// 设置TTL
int ttl = 64;
setsockopt(sockfd, IPPROTO_IP, IP_MULTICAST_TTL,
           &ttl, sizeof(ttl));

sendto(sockfd, data, len, 0,
       (struct sockaddr*)&mcast_addr, sizeof(mcast_addr));

// 2. 接收端
// 加入组播组
struct ip_mreq mreq;
inet_pton(AF_INET, "239.0.0.1", &mreq.imr_multiaddr);
mreq.imr_interface.s_addr = INADDR_ANY;

setsockopt(sockfd, IPPROTO_IP, IP_ADD_MEMBERSHIP,
           &mreq, sizeof(mreq));

// 绑定端口
bind(sockfd, INADDR_ANY, 8888);

// 接收组播数据
recvfrom(sockfd, buffer, len, ...);

// 离开组播组
setsockopt(sockfd, IPPROTO_IP, IP_DROP_MEMBERSHIP,
           &mreq, sizeof(mreq));
