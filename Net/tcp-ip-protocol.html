    2.1 IP协议详解

    2.1.1 IPv4报文格式

    /* IPv4头部结构 (RFC 791) - 20字节固定部分 */
struct iphdr {
    u8 version:4;      // 版本号(4)
    u8 ihl:4;          // 首部长度(单位4字节，最小5，最大15)
    u8 tos;              // 服务类型(DSCP + ECN)
    u16 tot_len;         // 总长度(含头部和数据，最大65535)
    u16 id;              // 标识(用于分片重组)
    u16 frag_off;        // 3bit标志 + 13bit片偏移
    u8 ttl;             // 生存时间(每经过路由器减1)
    u8 protocol;        // 上层协议(1=ICMP,6=TCP,17=UDP)
    u16 check;           // 头部校验和
    u32 saddr;           // 源IP地址(32位)
    u32 daddr;           // 目的IP地址(32位)
};

/* IP分片标志位 */
#define IP_RF 0x8000    // 保留标志
#define IP_DF 0x4000    // Don't Fragment
#define IP_MF 0x2000    // More Fragments
#define IP_OFFSET 0x1FFF// 片偏移掩码

    2.1.2 IP地址与子网划分

    类别 首字节范围 网络位/主机位 默认子网掩码 用途
    A类 1-126 8/24 255.0.0.0 大型网络
    B类 128-191 16/16 255.255.0.0 中型网络
    C类 192-223 24/8 255.255.255.0 小型网络
    D类 224-239 - - 组播
    E类 240-255 - - 保留

    2.1.3 ICMP协议

    /* ICMP报文格式 */
struct icmphdr {
    u8 type;            // ICMP类型
    u8 code;            // 代码
    u16 checksum;       // 校验和
    union {
        struct {
            u16 id;
            u16 sequence;
        } echo;          // Echo请求/应答
        u32 gateway;     // 重定向网关地址
        struct {
            u16 unused;
            u16 mtu;
        } frag;          // MTU发现
    } un;
};

/* 常见ICMP类型 */
#define ICMP_ECHOREPLY      0  // Echo应答(ping响应)
#define ICMP_DEST_UNREACH   3  // 目标不可达
#define ICMP_SOURCE_QUENCH  4  // 源抑制
#define ICMP_REDIRECT       5  // 重定向
#define ICMP_ECHO           8  // Echo请求(ping)
#define ICMP_TIME_EXCEEDED  11 // 超时(TTL=0)
#define ICMP_PARAMETERPROB  12 // 参数问题

    2.2 TCP协议深度剖析

    2.2.1 TCP报文格式

    /* TCP头部结构 (RFC 793) - 20-60字节 */
struct tcphdr {
    u16 source;          // 源端口(0-65535)
    u16 dest;            // 目的端口(0-65535)
    u32 seq;             // 序列号(ISN随机生成)
    u32 ack_seq;         // 确认号(下一个期望的序列号)
    u16 res1:4;        // 保留(0)
    u16 doff:4;        // 数据偏移(单位4字节，最小5)
    u16 fin:1;         // FIN: 结束连接
    u16 syn:1;         // SYN: 同步序列号
    u16 rst:1;         // RST: 重置连接
    u16 psh:1;         // PSH: 推送数据
    u16 ack:1;         // ACK: 确认号有效
    u16 urg:1;         // URG: 紧急指针有效
    u16 ece:1;         // ECE: ECN回显
    u16 cwr:1;         // CWR: 拥塞窗口减小
    u16 window;          // 窗口大小(0-65535，可窗口扩展至1GB)
    u16 check;           // 校验和(含伪头部)
    u16 urg_ptr;         // 紧急指针
};

/* TCP选项 (最多40字节) */
#define TCPOPT_EOL          0  // End of Option List
#define TCPOPT_NOP          1  // No Operation
#define TCPOPT_MAXSEG       2  // MSS(Maximum Segment Size)
#define TCPOPT_WINDOW       3  // 窗口扩展因子
#define TCPOPT_SACK_PERM    4  // SACK Permitted
#define TCPOPT_SACK         5  // SACK Block
#define TCPOPT_TIMESTAMP    8  // 时间戳

    2.2.2 TCP状态机

    /* TCP连接状态机 (11个状态) */
enum {
    TCP_ESTABLISHED = 1,  // 已建立连接
    TCP_SYN_SENT,         // 主动打开: 发送SYN，等待匹配的SYN
    TCP_SYN_RECV,         // 被动打开: 收到SYN并发送SYN+ACK
    TCP_FIN_WAIT1,        // 主动关闭: 发送FIN，等待ACK
    TCP_FIN_WAIT2,        // 主动关闭: 收到ACK，等待对方FIN
    TCP_TIME_WAIT,        // 2MSL等待状态(MSL=2分钟)
    TCP_CLOSE,            // 关闭状态
    TCP_CLOSE_WAIT,       // 被动关闭: 收到FIN，等待应用关闭
    TCP_LAST_ACK,         // 被动关闭: 发送FIN，等待ACK
    TCP_LISTEN,           // 监听状态
    TCP_CLOSING,          // 同时关闭: 等待ACK
};

/* 状态转换示例 */
// 主动打开: CLOSED -> SYN_SENT -> ESTABLISHED
// 被动打开: CLOSED -> LISTEN -> SYN_RECV -> ESTABLISHED
// 主动关闭: ESTABLISHED -> FIN_WAIT1 -> FIN_WAIT2 -> TIME_WAIT -> CLOSED
// 被动关闭: ESTABLISHED -> CLOSE_WAIT -> LAST_ACK -> CLOSED

    2.2.3 TCP可靠传输机制

    机制 原理 关键字段 作用
    序列号/确认号 每个字节有序列号 seq, ack_seq 保证顺序交付
    超时重传 未收到ACK则重传 RTO(重传超时) 应对丢包
    快速重传 3个重复ACK立即重传 duplicate ACK 快速恢复
    校验和 检测数据损坏 checksum 保证数据完整
    流量控制 滑动窗口 window 防止接收端溢出
    拥塞控制 动态调整发送速率 cwnd, ssthresh 防止网络拥塞

    2.2.4 TCP计时器

    /* TCP重要计时器 */

// 1. 重传计时器 (Retransmission Timer)
//    RTO计算：Jacobson算法
RTT = 测量的往返时间
SRTT = (1-α)*SRTT + α*RTT        // 平滑RTT (α=1/8)
RTTVAR = (1-β)*RTTVAR + β*|SRTT-RTT|  // RTT方差 (β=1/4)
RTO = SRTT + 4*RTTVAR                // 重传超时

// 2. 坚持计时器 (Persist Timer)
//    用于零窗口探测
if (recv_window == 0) {
    send_window_probe();  // 发送1字节探测
}

// 3. 保活计时器 (Keepalive Timer)
//    检测对方是否活着 (默认2小时)
if (idle_time > keepalive_time) {
    send_keepalive_probe();
}

// 4. TIME_WAIT计时器 (2MSL Timer)
//    2MSL = 2 * Maximum Segment Lifetime
//    Linux中 MSL = 60秒，2MSL = 120秒

    2.3 UDP协议

    /* UDP头部结构 (RFC 768) - 仅8有8字节 */
struct udphdr {
    u16 source;      // 源端口
    u16 dest;        // 目的端口
    u16 len;         // 长度(含头部，最小8，最大65535)
    u16 check;       // 校验和(可选)
};

/* UDP特点 */
// 1. 无连接：不需要建立连接，直接发送
// 2. 不可靠：不保证送达，不保证顺序
// 3. 面向报文：保持消息边界
// 4. 开销小：没有连接管理、流量控制等开销

/* UDP适用场景 */
// - DNS查询(快速响应)
// - 流媒体(RTSP/RTP)：容忍丢包
// - 在线游戏：低延迟优先
// - DHCP/TFTP：简单协议

    🛠️ 第三层：应用场景

    3.1 TCP三次握手

    /* 三次握手过程 */
// 第一次握手：客户端发送SYN
Client -> Server: SYN, seq=x

// 第二次握手：服务器发送SYN+ACK
Server -> Client: SYN+ACK, seq=y, ack=x+1

// 第三次握手：客户端发送ACK
Client -> Server: ACK, seq=x+1, ack=y+1

// 连接建立，进入ESTABLISHED状态
    💡 为什么需要三次握手？
    • 防止已失效的连接请求突然又传送到服务器
    • 确认双方的发送和接收能力
    • 协商初始序列号

    3.2 TCP四次挥手

    /* 四次挥手过程 */
// 第一次挥手：客户端发送FIN
Client -> Server: FIN, seq=u

// 第二次挥手：服务器发送ACK
Server -> Client: ACK, ack=u+1

// 第三次挥手：服务器发送FIN
Server -> Client: FIN, seq=w

// 第四次挥手：客户端发送ACK
Client -> Server: ACK, ack=w+1

// 客户端进入TIME_WAIT (2MSL)

    3.4.1 拥塞控制算法

    算法 阶段 特点
    慢启动 cwnd < ssthresh 指数增长
    拥塞避免 cwnd >= ssthresh 线性增长

    4.2 零拷贝技术

    /* sendfile系统调用 */
#include <sys/sendfile.h>

ssize_t sendfile(int out_fd, int in_fd,
                   off_t *offset, size_t count);

// 示例：高效文件传输
int send_file(int sockfd, const char* filename) {
    int fd = open(filename, O_RDONLY);
    struct stat st;
    fstat(fd, &st);
    
    // 直接从文件到socket，避免用户态拷贝
    sendfile(sockfd, fd, NULL, st.st_size);
    close(fd);
}
