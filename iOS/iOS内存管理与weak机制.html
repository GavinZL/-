🔗 三、weak弱引用原理

3.1 weak_table_t结构

objc4源码
struct weak_table_t {
    weak_entry_t *weak_entries;  // 弱引用条目数组
    size_t num_entries;          // 条目数量
    uintptr_t mask;              // 哈希表掩码
};

struct weak_entry_t {
    DisguisedPtr referent;  // 被弱引用的对象
    union {
        weak_referrer_t inline_referrers[4];  // 内联数组
        struct {
            weak_referrer_t *referrers;  // 动态数组
            uintptr_t num_refs;
        };
    };
};

3.2 weak注册流程

__weak赋值时
id objc_initWeak(id *location, id newObj) {
    *location = nil;
    return objc_storeWeak(location, newObj);
}

id objc_storeWeak(id *location, id newObj) {
    // 1. 获取SideTable
    SideTable *newTable = &SideTables()[newObj];
    
    // 2. 加锁
    newTable->lock();
    
    // 3. 在weak_table中查找或创建weak_entry
    weak_entry_t *entry = weak_entry_for_referent(&newTable->weak_table, newObj);
    
    if (!entry) {
        // 创建新条目
        weak_entry_t new_entry(newObj, location);
        weak_grow_maybe(&newTable->weak_table);
        weak_entry_insert(&newTable->weak_table, &new_entry);
    } else {
        // 添加到现有条目
        append_referrer(entry, location);
    }
    
    // 4. 标记对象被弱引用
    newObj->setWeaklyReferenced_nolock();
    
    // 5. 解锁并返回
    newTable->unlock();
    *location = newObj;
    return newObj;
}

3.3 weak清理流程

对象dealloc时
void weak_clear_no_lock(weak_table_t *weak_table, id referent) {
    // 1. 查找对象的weak_entry
    weak_entry_t *entry = weak_entry_for_referent(weak_table, referent);
    if (!entry) return;
    
    // 2. 遍历所有弱引用指针，置为nil
    weak_referrer_t *referrers;
    size_t count;
    
    if (entry->out_of_line()) {
        referrers = entry->referrers;
        count = entry->num_refs;
    } else {
        referrers = entry->inline_referrers;
        count = WEAK_INLINE_COUNT;
    }
    
    for (size_t i = 0; i < count; ++i) {
        id *referrer = referrers[i];
        if (referrer) {
            *referrer = nil;  // ⭐ 置为nil
        }
    }
    
    // 3. 移除weak_entry
    weak_entry_remove(weak_table, entry);
}
⚠️ weak工作原理总结
1. 注册阶段：weak指针赋值时，在weak_table中注册
2. 存储结构：对象地址→weak指针地址数组的映射
3. 自动置nil：对象dealloc时遍历数组，将所有weak指针置nil
4. 线程安全：通过SideTable的spinlock保证
5. 性能优化：小于4个弱引用用内联数组，大于4个用动态数组

3.4 weak vs unowned

特性weakunowned
对象释放后自动置nil不处理(野指针)
可选类型必须是Optional可以非Optional
性能较慢(需要查表)快(无额外操作)
安全性安全不安全(可能崩溃)
使用场景对象可能被释放确保对象存在

解决方案
// 方案1：使用weakProxy
@interface WeakProxy : NSProxy
@property (nonatomic, weak) id target;
@end

WeakProxy *proxy = [WeakProxy alloc];
proxy.target = self;
self.timer = [NSTimer scheduledTimerWithTimeInterval:1.0
                                              target:proxy
                                            selector:@selector(timerFired)
                                            userInfo:nil
                                             repeats:YES];

// 方案2：iOS 10+ block方式

    [self timerFired];
}];
