Swift 泛型详解

📌 文档概述
本文档全面讲解 Swift 中的泛型（Generics）机制，从基础概念到高级应用，结合源码分析和实战案例，帮助深入理解 Swift 类型系统的核心特性。

一、概念与背景

1.1 什么是泛型

泛型是一种编程范式，允许我们编写灵活、可重用的函数和类型，这些函数和类型可以与任何类型一起工作，同时保持类型安全。泛型是 Swift 最强大的特性之一，Swift 标准库大量使用了泛型。

核心价值

  • 代码复用：避免重复编写相似的代码
  • 类型安全：编译时确保类型正确，避免运行时错误
  • 抽象能力：提供更高层次的抽象，专注于算法逻辑而非具体类型
  • 性能优化：静态派发，无需运行时类型检查

1.2 泛型的历史背景

泛型编程起源于 C++ 的模板（Templates），后来被 Java、C# 等语言采纳。Swift 的泛型设计借鉴了这些语言的优点，同时结合了协议导向编程的理念，形成了更加强大和优雅的类型系统。

// 没有泛型的世界：需要为每种类型编写重复代码
func swapTwoInts(_ a: inout Int, _ b: inout Int) {
    let temp = a
    a = b
    b = temp
}

func swapTwoStrings(_ a: inout String, _ b: inout String) {
    let temp = a
    a = b
    b = temp
}

// 使用泛型：一次编写，到处使用
func swapTwoValues<T>(_ a: inout T, _ b: inout T) {
    let temp = a
    a = b
    b = temp
}

2.1 泛型函数

2.1.2 多类型参数

// 多个类型参数的泛型函数
func zip<T, U>(_ first: T, _ second: U) -> (T, U) {
    return (first, second)
}

let result = zip(42, "Answer")
// result: (Int, String) = (42, "Answer")

// 更复杂的示例：字典转换
func transformDictionary<Key, Value, TransformedValue>(
    _ dictionary: [Key: Value],
    transform: (Value) -> TransformedValue
) -> [Key: TransformedValue] {
    var result: [Key: TransformedValue] = [:]
    for (key, value) in dictionary {
        result[key] = transform(value)
    }
    return result
}

let grades = transformDictionary(scores) { score in

2.2 泛型类型

2.2.1 泛型结构体

// 经典示例：栈数据结构
struct Stack<Element> {
    private var items: [Element] = []
    
    mutating func push(_ item: Element) {
        items.append(item)
    }
    
    mutating func pop() -> Element? {
        return items.isEmpty ? nil : items.removeLast()
    }
    
    func peek() -> Element? {
        return items.last
    }
    
    var isEmpty: Bool {
        return items.isEmpty
    }
    
    var count: Int {
        return items.count
    }
}

// 使用示例

var stringStack = Stack<String>()

stringStack.push("Generics")
print(stringStack.peek()) // Optional("Generics")

2.2.2 泛型类

// 泛型链表节点
class LinkedListNode<T> {
    var value: T
    var next: LinkedListNode<T>?
    
    init(value: T) {
        self.value = value
    }
}

// 泛型链表
class LinkedList<T> {
    private var head: LinkedListNode<T>?
    private var tail: LinkedListNode<T>?
    private(set) var count: Int = 0
    
    var isEmpty: Bool {
        return head == nil
    }
    
    func append(_ value: T) {
        let newNode = LinkedListNode(value: value)
        
        if let tail = tail {
            tail.next = newNode
            self.tail = newNode
        } else {
            head = newNode
            tail = newNode
        }
        
        count += 1
    }
    
    func removeFirst() -> T? {
        guard let first = head else { return nil }
        
        head = first.next
        if head == nil {
            tail = nil
        }
        
        count -= 1
        return first.value
    }
}

// 使用示例
let list = LinkedList<String>()
list.append("First")
list.append("Second")
list.append("Third")
print(list.removeFirst()) // Optional("First")

2.2.3 泛型枚举

// Swift 标准库中的 Optional 就是泛型枚举
enum Optional<Wrapped> {
    case none
    case some(Wrapped)
}

// 自定义泛型枚举：Result 类型
enum Result<Success, Failure: Error> {
    case success(Success)
    case failure(Failure)
    
    func map<U>(_ transform: (Success) -> U) -> Result<U, Failure> {
        switch self {
        case .success(let value):
            return .success(transform(value))
        case .failure(let error):
            return .failure(error)
        }
    }
}

// 更复杂的示例：二叉树
indirect enum BinaryTree<T> {
    case empty
    case node(left: BinaryTree<T>, value: T, right: BinaryTree<T>)
    
    func map<U>(_ transform: (T) -> U) -> BinaryTree<U> {
        switch self {
        case .empty:
            return .empty
        case .node(let left, let value, let right):
            return .node(
                left: left.map(transform),
                value: transform(value),
                right: right.map(transform)
            )
        }
    }
}

// 使用示例

    left: .node(left: .empty, value: 1, right: .empty),
    value: 2,

2.3 类型约束（Type Constraints）

2.3.1 协议约束

// 要求泛型类型遵循 Equatable 协议
func findIndex<T: Equatable>(of valueToFind: T, in array: [T]) -> Int? {
    for (index, value) in array.enumerated() {
        if value == valueToFind {
            return index
        }
    }
    return nil
}

let strings = ["apple", "banana", "cherry"]
if let index = findIndex(of: "banana", in: strings) {
    print("Found at index \(index)") // Found at index 1
}

// 多个协议约束
func sortedKeys<T: Hashable & Comparable, U>(_ dictionary: [T: U]) -> [T] {
    return dictionary.keys.sorted()
}

let ages = ["Alice": 25, "Bob": 30, "Charlie": 22]
let sorted = sortedKeys(ages)
// sorted: ["Alice", "Bob", "Charlie"]

2.4 关联类型（Associated Types）

关联类型是协议中的泛型占位符，允许协议定义时不指定具体类型，由遵循协议的类型来确定。

// 定义包含关联类型的协议
protocol Container {
    associatedtype Item
    
    mutating func append(_ item: Item)
    var count: Int { get }
    subscript(i: Int) -> Item { get }
}

// 实现协议
struct IntStack: Container {
    private var items: [Int] = []
    
    mutating func push(_ item: Int) {
        items.append(item)
    }
    
    // Container 协议实现
    typealias Item = Int
    
    mutating func append(_ item: Int) {
        push(item)
    }
    
    var count: Int {
        return items.count
    }
    
    subscript(i: Int) -> Int {
        return items[i]
    }
}

// 泛型类型实现协议
struct GenericStack<Element>: Container {
    private var items: [Element] = []
    
    mutating func append(_ item: Element) {
        items.append(item)
    }
    
    var count: Int {
        return items.count
    }
    
    subscript(i: Int) -> Element {
        return items[i]
    }
}

2.4.1 关联类型约束

// 关联类型可以有约束
protocol SortableContainer {
    associatedtype Item: Comparable
    
    mutating func append(_ item: Item)
    mutating func sort()
    var items: [Item] { get }
}

struct SortableStack<T: Comparable>: SortableContainer {
    private(set) var items: [T] = []
    
    mutating func append(_ item: T) {
        items.append(item)
    }
    
    mutating func sort() {
        items.sort()
    }
}

var numbers = SortableStack<Int>()
numbers.append(5)
numbers.append(2)
numbers.append(8)
numbers.sort()
print(numbers.items) // [2, 5, 8]

2.5 泛型 where 子句

where 子句为关联类型定义约束，也可以要求特定类型参数之间的关系。

// 要求两个容器的元素类型相同且可比较
func allItemsMatch<C1: Container, C2: Container>(
    _ container1: C1,
    _ container2: C2
) -> Bool where C1.Item == C2.Item, C1.Item: Equatable {
    if container1.count != container2.count {
        return false
    }
    
    for i in 0..<container1.count {
        if container1[i] != container2[i] {
            return false
        }
    }
    
    return true
}

// 扩展中使用 where 子句
extension Array where Element: Numeric {
    func sum() -> Element {
        return reduce(0, +)
    }
    
    func average() -> Double where Element == Int {
        guard !isEmpty else { return 0 }
        return Double(sum()) / Double(count)
    }
}

let numbers = [1, 2, 3, 4, 5]
print(numbers.sum()) // 15
print(numbers.average()) // 3.0

// 更复杂的 where 子句示例
extension Container where Item: Equatable {
    func contains(_ item: Item) -> Bool {
        for i in 0..<count {
            if self[i] == item {
                return true
            }
        }
        return false
    }
}

2.6 不透明类型（Opaque Types）- some 关键字

Swift 5.1 引入的 some 关键字，用于隐藏具体类型，只暴露协议接口。

// 返回不透明类型
protocol Shape {
    func draw() -> String
}

struct Circle: Shape {
    var radius: Double
    
    func draw() -> String {
        return "Circle with radius \(radius)"
    }
}

struct Rectangle: Shape {
    var width: Double
    var height: Double
    
    func draw() -> String {
        return "Rectangle \(width)x\(height)"
    }
}

// 使用 some 返回不透明类型
func makeShape() -> some Shape {
    return Circle(radius: 5.0)
}

let shape = makeShape()
print(shape.draw()) // Circle with radius 5.0

// SwiftUI 中的典型应用
protocol View {
    associatedtype Body: View
    var body: Body { get }
}

struct ContentView: View {
    var body: some View {
        // 具体类型被隐藏
        Text("Hello, World!")
    }
}
💡 some vs any（Swift 5.7+）
  • some Protocol：不透明类型，编译器知道具体类型，类型固定，性能好
  • any Protocol：存在类型，可以存储不同的遵循协议的类型，更灵活但有性能开销
// some: 类型固定
func makeSomeShape() -> some Shape {
    return Circle(radius: 5.0)
    // 始终返回相同类型
}

// any: 类型可变
func makeAnyShape(isCircle: Bool) -> any Shape {
    if isCircle {
        return Circle(radius: 5.0)
    } else {
        return Rectangle(width: 10, height: 20)
    }
}

// any 可以存储不同类型
var shapes: [any Shape] = []
shapes.append(Circle(radius: 3.0))
shapes.append(Rectangle(width: 5, height: 10))

2.7 泛型的底层实现原理

2.7.1 特化（Specialization）

Swift 编译器会对泛型代码进行特化优化，为每个具体类型生成专门的代码，避免运行时类型检查。

// 源代码
func swap<T>(_ a: inout T, _ b: inout T) {
    let temp = a
    a = b
    b = temp
}

// 编译器生成的特化版本（概念示例）
// 对于 Int
func swap_Int(_ a: inout Int, _ b: inout Int) {
    let temp = a
    a = b
    b = temp
}

// 对于 String
func swap_String(_ a: inout String, _ b: inout String) {
    let temp = a
    a = b
    b = temp
}

2.7.2 见证表（Witness Table）

当泛型有协议约束时，Swift 使用见证表来实现动态派发。

// 协议见证表示例
protocol Drawable {
    func draw()
}

struct Point: Drawable {

// 见证表结构（概念）

3.1 集合类型

// Swift 标准库中的集合都是泛型
let array: [Int] = [1, 2, 3]
let dictionary: [String: Int] = ["one": 1, "two": 2]

// 自定义泛型集合：优先队列
struct PriorityQueue<Element: Comparable> {
    private var heap: [Element] = []
    
    var isEmpty: Bool {
        return heap.isEmpty
    }
    
    var peek: Element? {
        return heap.first
    }
    
    mutating func enqueue(_ element: Element) {
        heap.append(element)
        siftUp(from: heap.count - 1)
    }
    
    mutating func dequeue() -> Element? {
        guard !heap.isEmpty else { return nil }
        
        if heap.count == 1 {
            return heap.removeLast()
        }
        
        let value = heap[0]
        heap[0] = heap.removeLast()
        siftDown(from: 0)
        return value
    }
    
    private mutating func siftUp(from index: Int) {
        var child = index
        var parent = (child - 1) / 2
        
        while child > 0 && heap[child] > heap[parent] {
            heap.swapAt(child, parent)
            child = parent
            parent = (child - 1) / 2
        }
    }
    
    private mutating func siftDown(from index: Int) {
        var parent = index
        
        while true {
            let left = 2 * parent + 1
            let right = 2 * parent + 2
            var candidate = parent
            
            if left < heap.count && heap[left] > heap[candidate] {
                candidate = left
            }
            
            if right < heap.count && heap[right] > heap[candidate] {
                candidate = right
            }
            
            if candidate == parent {
                return
            }
            
            heap.swapAt(parent, candidate)
            parent = candidate
        }
    }
}

// 使用示例
var pq = PriorityQueue<Int>()
pq.enqueue(5)
pq.enqueue(10)
pq.enqueue(3)
pq.enqueue(8)

while let max = pq.dequeue() {
    print(max) // 10, 8, 5, 3
}

3.2 函数式编程

// map, flatMap, filter 等都是泛型方法
extension Array {
    func customMap<T>(_ transform: (Element) -> T) -> [T] {
        var result: [T] = []
        for element in self {
            result.append(transform(element))
        }
        return result
    }
    
    func customFilter(_ predicate: (Element) -> Bool) -> [Element] {
        var result: [Element] = []
        for element in self {
            if predicate(element) {
                result.append(element)
            }
        }
        return result
    }
    
    func customReduce<Result>(
        _ initialResult: Result,
        _ nextPartialResult: (Result, Element) -> Result
    ) -> Result {
        var result = initialResult
        for element in self {
            result = nextPartialResult(result, element)
        }
        return result
    }
}

// 链式调用示例
let numbers = [1, 2, 3, 4, 5, 6]
let result = numbers
    .customFilter { $0 % 2 == 0 }
    .customMap { $0 * $0 }
    .customReduce(0, +)

print(result) // 56 (4 + 16 + 36)

3.4 数据持久化

// 泛型存储协议
protocol StorageService {
    func save<T: Encodable>(_ value: T, forKey key: String) throws
    func load<T: Decodable>(forKey key: String) throws -> T?
    func remove(forKey key: String)
}

// UserDefaults 实现
class UserDefaultsStorage: StorageService {
    private let userDefaults: UserDefaults
    
    init(userDefaults: UserDefaults = .standard) {
        self.userDefaults = userDefaults
    }
    
    func save<T: Encodable>(_ value: T, forKey key: String) throws {
        let data = try JSONEncoder().encode(value)
        userDefaults.set(data, forKey: key)
    }
    
    func load<T: Decodable>(forKey key: String) throws -> T? {
        guard let data = userDefaults.data(forKey: key) else {
            return nil
        }
        return try JSONDecoder().decode(T.self, from: data)
    }
    
    func remove(forKey key: String) {
        userDefaults.removeObject(forKey: key)
    }
}

// 使用示例
struct Settings: Codable {
    var theme: String
    var fontSize: Int
}

let storage = UserDefaultsStorage()
let settings = Settings(theme: "dark", fontSize: 14)

try? storage.save(settings, forKey: "app_settings")
let loaded: Settings? = try? storage.load(forKey: "app_settings")

3.5 状态管理

// 泛型状态管理
class Observable<T> {
    typealias Observer = (T) -> Void
    
    private var observers: [Observer] = []
    
    var value: T {
        didSet {
            notifyObservers()
        }
    }
    
    init(_ value: T) {
        self.value = value
    }
    
    func observe(_ observer: @escaping Observer) {
        observers.append(observer)
        observer(value)
    }
    
    private func notifyObservers() {
        observers.forEach { $0(value) }
    }
}

// 使用示例
let counter = Observable(0)

counter.observe { value in
    print("Counter changed to: \(value)")
}

counter.value = 10 // Counter changed to: 10
counter.value = 20 // Counter changed to: 20

四、高级技巧

4.1 类型擦除（Type Erasure）

当需要存储不同泛型参数的同一协议类型时，使用类型擦除技术。

// 问题：不能直接创建协议数组
protocol Vehicle {
    associatedtype FuelType
    func refuel(with fuel: FuelType)
}

// var vehicles: [Vehicle] = [] // ❌ 编译错误

// 解决方案：类型擦除
class AnyVehicle<FuelType>: Vehicle {
    private let _refuel: (FuelType) -> Void
    
    init<V: Vehicle>(_ vehicle: V) where V.FuelType == FuelType {
        _refuel = vehicle.refuel
    }
    
    func refuel(with fuel: FuelType) {
        _refuel(fuel)
    }
}

// 具体实现
struct Car: Vehicle {
    func refuel(with fuel: String) {
        print("Refueling car with \(fuel)")
    }
}

struct ElectricCar: Vehicle {
    func refuel(with fuel: Int) {
        print("Charging electric car with \(fuel) kWh")
    }
}

// 现在可以创建数组
let gasolineVehicles: [AnyVehicle<String>] = [
    AnyVehicle(Car())
]

// Swift 标准库中的类型擦除示例
let anySequence = AnySequence([1, 2, 3])
let anyCollection = AnyCollection(["a", "b", "c"])

4.2 泛型递归约束

// 自引用协议
protocol Copyable {
    func copy() -> Self
}

class Node<T>: Copyable {
    var value: T
    var next: Node<T>?
    
    init(value: T) {
        self.value = value
    }
    
    func copy() -> Self {
        let newNode = Node(value: value)
        newNode.next = next?.copy()
        return newNode as! Self
    }
}

// 递归泛型约束
protocol TreeNode {
    associatedtype Value
    associatedtype Child: TreeNode where Child.Value == Value
    
    var value: Value { get }
    var children: [Child] { get }
}

4.3 幻影类型（Phantom Types）

// 使用泛型参数来编码额外的类型信息，但不实际存储该类型的值
struct Validated<T> { }

enum Valid { }
enum Invalid { }

struct Email<State = Validated<Invalid>> {
    let value: String
    
    init(_ value: String) {
        self.value = value
    }
}

extension Email where State == Validated<Invalid> {
    func validate() -> Email<Validated<Valid>>? {
        guard value.contains("@") else { return nil }
        return Email<Validated<Valid>>(value)
    }
}

extension Email where State == Validated<Valid> {
    func send(message: String) {
        print("Sending '\(message)' to \(value)")
    }
}

// 使用示例
let email = Email("user@example.com")
// email.send(message: "Hello") // ❌ 编译错误：未验证的邮箱不能发送

if let validEmail = email.validate() {
    validEmail.send(message: "Hello") // ✅ OK
}

4.4 条件遵循（Conditional Conformance）

// 当泛型参数满足特定条件时，类型才遵循协议
extension Array: Equatable where Element: Equatable {
    // Array 在元素遵循 Equatable 时自动遵循 Equatable
}

extension Optional: Equatable where Wrapped: Equatable {
    // Optional 在包装类型遵循 Equatable 时自动遵循 Equatable
}

// 自定义示例
struct Pair<T, U> {
    let first: T
    let second: U
}

extension Pair: Equatable where T: Equatable, U: Equatable {
    static func == (lhs: Pair, rhs: Pair) -> Bool {
        return lhs.first == rhs.first && lhs.second == rhs.second
    }
}

extension Pair: Comparable where T: Comparable, U: Comparable {
    static func < (lhs: Pair, rhs: Pair) -> Bool {
        if lhs.first != rhs.first {
            return lhs.first < rhs.first
        }
        return lhs.second < rhs.second
    }
}

// 使用示例
let pair1 = Pair(first: 1, second: "a")
let pair2 = Pair(first: 1, second: "b")
print(pair1 < pair2) // true

4.5 高阶泛型（Higher-Kinded Types 的模拟）

// Swift 不直接支持高阶类型，但可以通过协议模拟
protocol Functor {
    associatedtype A
    associatedtype FB: Functor
    
    func map<B>(_ transform: (A) -> B) -> FB
}

protocol Monad: Functor {
    associatedtype MB: Monad
    
    static func pure(_ value: A) -> Self
    func flatMap<B>(_ transform: (A) -> MB) -> MB
}

// Optional 实现 Monad（概念示例）
extension Optional {
    static func pure(_ value: Wrapped) -> Optional<Wrapped> {
        return .some(value)
    }
}

4.6 泛型性能优化

性能最佳实践

  • 使用值类型：struct 和 enum 配合泛型性能更好
  • 避免装箱：使用 some 而不是 any
  • 编译器优化：使用 @inlinable 和 @_specialize
  • 减少协议约束：过多约束会增加编译时间
// 使用 @inlinable 允许跨模块优化
@inlinable
public func swap<T>(_ a: inout T, _ b: inout T) {
    let temp = a
    a = b
    b = temp
}

// 使用 @_specialize 为特定类型生成优化代码
@_specialize(where T == Int)
@_specialize(where T == String)
func process<T: Equatable>(_ items: [T]) -> [T] {
    // 处理逻辑
    return items
}

// 使用 some 避免存在类型的性能开销
func makeCollection() -> some Collection<Int> {
    return [1, 2, 3] // 具体类型被隐藏但编译器知道
}

// 而不是使用 any（有运行时开销）
func makeAnyCollection() -> any Collection<Int> {
    return [1, 2, 3] // 使用存在容器，有性能开销
}

五、常见问题与最佳实践

5.1 常见问题

⚠️ 问题1：Protocol can only be used as a generic constraint
当协议包含关联类型或 Self 要求时，不能直接作为类型使用。 
// ❌ 错误
protocol Container {
    associatedtype Item
}

// var container: Container // ❌ 编译错误

// ✅ 解决方案 1：使用泛型
func process<C: Container>(container: C) {
    // ...
}

// ✅ 解决方案 2：使用 any
var container: any Container

// ✅ 解决方案 3：类型擦除
struct AnyContainer<Item>: Container {
    // ...
}
⚠️ 问题2：Cannot convert return expression of type
泛型函数返回类型不匹配。 
// ❌ 错误
func makeArray<T>() -> [T] {
    return [] // T 没有被推断
}

// ✅ 解决方案：提供类型信息
func makeArray<T>(of type: T.Type) -> [T] {
    return []
}

let ints = makeArray(of: Int.self)

// 或者通过参数推断
func makeArray<T>(containing element: T) -> [T] {
    return [element]
}

5.2 最佳实践

场景 建议 原因
返回类型 优先使用 some Protocol 更好的性能和类型推断
存储属性 使用具体泛型类型或 any Protocol some 不能用于存储属性
集合元素 使用 any Protocol 或类型擦除 允许存储不同具体类型
性能关键代码 使用值类型 + 泛型 编译器优化更好
公共 API 使用 @inlinable 允许跨模块优化

5.3 设计原则

📐 SOLID 原则在泛型中的应用
  1. 单一职责：每个泛型类型应该有清晰的单一用途
  2. 开闭原则：通过泛型和协议扩展，对扩展开放，对修改关闭
  3. 里氏替换：泛型约束确保类型替换的安全性
  4. 接口隔离：使用多个小协议而不是大而全的协议
  5. 依赖倒置：依赖抽象（协议），而不是具体类型
// 好的设计：小而专注的协议
protocol Readable {
    func read() -> String
}

protocol Writable {
    func write(_ content: String)
}

// 组合使用
func copyContent<Source: Readable, Destination: Writable>(
    from source: Source,
    to destination: Destination
) {
    let content = source.read()
    destination.write(content)
}

// 而不是：
protocol FileSystem {
    func read() -> String
    func write(_ content: String)
    func delete()
    func rename(_ newName: String)
    // 太多职责
}
